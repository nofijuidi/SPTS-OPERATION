async function executeSubmission() { const type = document.getElementById('equipmentTypeSelect').value; const num = document.getElementById('equipmentNumSelect').value; const btnConfirm = document.getElementById('btnConfirmSubmit'); btnConfirm.disabled = true; btnConfirm.innerText = "Submitting..."; const defectResults = []; let hasDefects = false; // Iterate over checklist items and skip any item marked as 'G' (Good) Object.values(checklistItemStates).forEach(st => { if (st.status !== 'G') { hasDefects = true; const cleanTitle = getCleanItemTitle(st.itemTitle); const statusText = STATUS_MAP[st.status] || st.status; const remarkText = st.remark ? `, ${st.remark}` : ''; defectResults.push(`${cleanTitle} - ${statusText}${remarkText}`); } }); // If defects exist, join only the defects. Otherwise set string to "All is Good" const areaString = hasDefects ? defectResults.join(' | ') : "All is Good"; const now = new Date(); const year = now.getFullYear(); const month = String(now.getMonth() + 1).padStart(2, '0'); const day = String(now.getDate()).padStart(2, '0'); const hours = String(now.getHours()).padStart(2, '0'); const minutes = String(now.getMinutes()).padStart(2, '0'); const seconds = String(now.getSeconds()).padStart(2, '0'); const isoDateTimeString = `${year}-${month}-${day}T${hours}:${minutes}:${seconds}`; const currentShift = document.getElementById('shiftNameTag').innerText; const payload = { date: isoDateTimeString, shift: currentShift, COMPANY_ID: getPropValue(currentUser, ['COMPANY_ID', 'company_id', 'Company_ID', 'ID', 'id', 'Username'], ''), Title: getPropValue(currentUser, ['NAME', 'Title', 'title', 'name'], ''), company: getPropValue(currentUser, ['COMPANY', 'Company', 'company'], 'SPTS'), group: getPropValue(currentUser, ['GROUP', 'Group', 'group'], ''), equipment: type, equipnum: num, area: areaString, declaration: hasDefects ? "Reported with Defects" : "Declared All OK", action: "New Inspection Log", isNewIssue: hasDefects }; try { const response = await fetch(INSPECTION_SUBMIT_FLOW_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); if (response.ok) { alert("Inspection checklist submitted successfully!"); closeConfirmationModal(); goHome(); } else { alert("Failed to submit inspection checklist."); } } catch (err) { console.error("Submission Error:", err); alert("Network error submitting checklist."); } finally { btnConfirm.disabled = false; btnConfirm.innerText = "Confirm"; } }
